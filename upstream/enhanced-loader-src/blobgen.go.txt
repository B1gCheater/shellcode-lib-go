//go:build ignore

// blobgen regenerates donut/zz_loader_blobs_enhanced.go from two raw .text blobs.
// Usage: go run main.go <x86.bin> <x64.bin> <output.go>
package main

import (
	"fmt"
	"os"
)

func emitArray(f *os.File, name, comment string, blob []byte) {
	fmt.Fprintf(f, "// %s\n", comment)
	fmt.Fprintf(f, "var %s = []byte{\n", name)
	for i, b := range blob {
		if i%12 == 0 {
			fmt.Fprintf(f, "\t")
		}
		fmt.Fprintf(f, "0x%02x, ", b)
		if i%12 == 11 {
			fmt.Fprintf(f, "\n")
		}
	}
	if len(blob)%12 != 0 {
		fmt.Fprintf(f, "\n")
	}
	fmt.Fprintf(f, "}\n\n")
}

func main() {
	if len(os.Args) != 4 {
		fmt.Fprintln(os.Stderr, "usage: blobgen <x86.bin> <x64.bin> <output.go>")
		os.Exit(2)
	}
	x86, err := os.ReadFile(os.Args[1])
	must(err)
	x64, err := os.ReadFile(os.Args[2])
	must(err)

	f, err := os.Create(os.Args[3])
	must(err)
	defer f.Close()

	fmt.Fprintf(f, "// Code generated by blobgen. DO NOT EDIT.\n\n")
	fmt.Fprintf(f, "package donut\n\n")
	emitArray(f, "loaderEnhancedX86",
		"loaderEnhancedX86 holds the x86 (32-bit) loader stub.\n//\n// Generated by blobgen from donut's loader header; do not edit.", x86)
	emitArray(f, "loaderEnhancedX64",
		"loaderEnhancedX64 holds the amd64 (64-bit) loader stub.\n//\n// Generated by blobgen from donut's loader header; do not edit.", x64)
	fmt.Printf("wrote %s (x86=%d bytes, x64=%d bytes)\n", os.Args[3], len(x86), len(x64))
}

func must(err error) {
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
